## 📊 数据模型设计

### 1. 用药计划表

```sql
CREATE TABLE med_plans (
    id BIGINT NOT NULL AUTO_INCREMENT COMMENT '计划ID',
    patient_id BIGINT NOT NULL COMMENT '患者ID',
    plan_name VARCHAR(128) NOT NULL COMMENT '计划名称',
    start_date DATE NOT NULL COMMENT '开始日期',
    end_date DATE COMMENT '结束日期',
    status TINYINT NOT NULL DEFAULT 1 COMMENT '状态(1-活跃,2-暂停,3-终止)',
    version INT NOT NULL DEFAULT 1 COMMENT '版本号',
    description TEXT COMMENT '计划描述',
    instructions TEXT COMMENT '用药说明',
    creator VARCHAR(64) DEFAULT '' COMMENT '创建者',
    create_time DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    updater VARCHAR(64) DEFAULT '' COMMENT '更新者',
    update_time DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    deleted BIT(1) NOT NULL DEFAULT b'0' COMMENT '是否删除',
    tenant_id BIGINT NOT NULL DEFAULT '0' COMMENT '租户编号',
    PRIMARY KEY (id),
    KEY idx_med_plan_patient_id (patient_id),
    KEY idx_med_plan_status (status),
    KEY idx_med_plan_start_date (start_date)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='用药计划表';
```

## 🔧 核心功能实现

### 1. 用药计划服务

#### 计划创建服务
```java
@Service
public class MedPlanServiceImpl implements MedPlanService {
    
    @Autowired
    private MedPlanMapper medPlanMapper;
    
    @Autowired
    private MedPlanItemMapper medPlanItemMapper;
    
    @Autowired
    private TodayTaskService todayTaskService;
    
    /**
     * 创建用药计划
     */
    @Transactional
    public Long createMedPlan(MedPlanCreateReqVO createReqVO) {
        // 1. 创建用药计划
        MedPlanDO medPlan = MedPlanConvert.INSTANCE.convert(createReqVO);
        medPlan.setStatus(MedPlanStatus.ACTIVE);
        medPlan.setVersion(1);
        medPlanMapper.insert(medPlan);
        
        // 2. 创建用药计划项
        for (MedPlanItemCreateReqVO itemReqVO : createReqVO.getItems()) {
            MedPlanItemDO planItem = MedPlanItemConvert.INSTANCE.convert(itemReqVO);
            planItem.setPlanId(medPlan.getId());
            medPlanItemMapper.insert(planItem);
        }
        
        // 3. 生成今日待办任务
        todayTaskService.generateTodayTasks(medPlan.getId());
        
        return medPlan.getId();
    }
    
    /**
     * 更新用药计划
     */
    @Transactional
    public void updateMedPlan(MedPlanUpdateReqVO updateReqVO) {
        // 1. 校验计划存在
        MedPlanDO existingPlan = medPlanMapper.selectById(updateReqVO.getId());
        if (existingPlan == null) {
            throw new ServiceException("用药计划不存在");
        }
        
        // 2. 更新计划信息
        MedPlanDO medPlan = MedPlanConvert.INSTANCE.convert(updateReqVO);
        medPlan.setVersion(existingPlan.getVersion() + 1);
        medPlanMapper.updateById(medPlan);
        
        // 3. 更新计划项
        updateMedPlanItems(medPlan.getId(), updateReqVO.getItems());
        
        // 4. 重新生成今日待办任务
        todayTaskService.regenerateTodayTasks(medPlan.getId());
    }
}
```

### 2. 今日待办服务

#### 待办任务生成器
```java
@Service
public class TodayTaskServiceImpl implements TodayTaskService {
    
    @Autowired
    private TodayTaskMapper todayTaskMapper;
    
    @Autowired
    private MedPlanItemMapper medPlanItemMapper;
    
    /**
     * 生成今日待办任务
     */
    public void generateTodayTasks(Long planId) {
        // 1. 获取计划项
        List<MedPlanItemDO> planItems = medPlanItemMapper.selectByPlanId(planId);
        
        // 2. 生成今日任务
        LocalDate today = LocalDate.now();
        for (MedPlanItemDO planItem : planItems) {
            generateTasksForPlanItem(planItem, today);
        }
    }
    
    /**
     * 为计划项生成任务
     */
    private void generateTasksForPlanItem(MedPlanItemDO planItem, LocalDate taskDate) {
        // 解析时间点
        List<LocalTime> timePoints = parseTimePoints(planItem.getTimePoints());
        
        // 为每个时间点生成任务
        for (LocalTime timePoint : timePoints) {
            TodayTaskDO task = TodayTaskDO.builder()
                .planItemId(planItem.getId())
                .patientId(planItem.getPatientId())
                .taskDate(taskDate)
                .scheduledTime(timePoint)
                .status(TaskStatus.PENDING)
                .build();
            
            // 检查任务是否已存在
            if (!isTaskExists(planItem.getId(), taskDate, timePoint)) {
                todayTaskMapper.insert(task);
            }
        }
    }
    
    /**
     * 解析时间点
     */
    private List<LocalTime> parseTimePoints(String timePointsJson) {
        try {
            ObjectMapper mapper = new ObjectMapper();
            return mapper.readValue(timePointsJson, 
                mapper.getTypeFactory().constructCollectionType(List.class, LocalTime.class));
        } catch (Exception e) {
            log.error("解析时间点失败: {}", timePointsJson, e);
            return new ArrayList<>();
        }
    }
    
    /**
     * 检查任务是否已存在
     */
    private boolean isTaskExists(Long planItemId, LocalDate taskDate, LocalTime scheduledTime) {
        return todayTaskMapper.selectByPlanItemAndDateTime(planItemId, taskDate, scheduledTime) != null;
    }
}
```

### 3. 计划状态管理

#### 计划状态操作
```java
@Service
public class MedPlanStatusService {
    
    /**
     * 暂停计划
     */
    @Transactional
    public void pauseMedPlan(Long planId) {
        MedPlanDO medPlan = medPlanMapper.selectById(planId);
        if (medPlan == null) {
            throw new ServiceException("用药计划不存在");
        }
        
        medPlan.setStatus(MedPlanStatus.PAUSED);
        medPlanMapper.updateById(medPlan);
        
        // 暂停相关任务
        todayTaskService.pauseTasksByPlanId(planId);
    }
    
    /**
     * 恢复计划
     */
    @Transactional
    public void resumeMedPlan(Long planId) {
        MedPlanDO medPlan = medPlanMapper.selectById(planId);
        if (medPlan == null) {
            throw new ServiceException("用药计划不存在");
        }
        
        medPlan.setStatus(MedPlanStatus.ACTIVE);
        medPlanMapper.updateById(medPlan);
        
        // 恢复相关任务
        todayTaskService.resumeTasksByPlanId(planId);
    }
    
    /**
     * 终止计划
     */
    @Transactional
    public void terminateMedPlan(Long planId) {
        MedPlanDO medPlan = medPlanMapper.selectById(planId);
        if (medPlan == null) {
            throw new ServiceException("用药计划不存在");
        }
        
        medPlan.setStatus(MedPlanStatus.TERMINATED);
        medPlanMapper.updateById(medPlan);
        
        // 终止相关任务
        todayTaskService.terminateTasksByPlanId(planId);
    }
}
```
